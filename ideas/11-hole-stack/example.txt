m6_clear_d = 6.71; // clearance hole, H13 band centre: clearance_medium_mm 6.6 [likely] (ISO 273:1979 medium series) + IT13/2 = 0.11 (fits.json 6-10 mm, IT13 220 um) [likely]; limits 6.600..6.820
m6_cbore_d = 10.75; // counterbore, band centre: shcs_head_d_knurled_mm 10.22 [likely] (ISO 4762:2004 Table 1, dk max (knurled head)) + 0.4 (table rule, columns.cbore_d_mm) + FLIGHT_PROCESSES.cnc_mill.tol_general_mm 0.13 [tool default]; limits 10.620..10.880

// not used: cbore_d_mm 11 [recall], the DIN 974-1 style counterbore tool size
// --- end parameters ---
function fn_tol(d, tol=tess_tol) = max(24, ceil(180 / acos(1 - min(0.5, 2*tol/d))));

// Clamped plate: every cut the M6 screw needs, as one solid to subtract.
module hole_stack_m6_clamp(){
  union(){
    translate([0, 0, m6_plate_t - m6_cbore_depth]) cylinder(d = m6_cbore_d, h = m6_cbore_depth + cut_ext, $fn = fn_tol(m6_cbore_d));
    translate([0, 0, -cut_ext]) cylinder(d = m6_clear_d, h = m6_plate_t + 2*cut_ext, $fn = fn_tol(m6_clear_d));
  }
}
